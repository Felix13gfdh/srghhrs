#include <iostream>

int main() {
    int x = 42; // Обычная переменная с числом

    int *p1 = &x;     // 1-й уровень (указатель на int)
    int **p2 = &p1;   // 2-й уровень (указатель на p1)
    int ***p3 = &p2;  // 3-й уровень (указатель на p2)
    int ****p4 = &p3; // 4-й уровень (указатель на p3)
    int *****p5 = &p4;// 5-й уровень (пятерной указатель на p4)

    // Чтобы добраться до изначального числа, нужно использовать 5 звездочек (*):
    std::cout << "*****p5 = " << *****p5 << std::endl;

    return 0;
}
